# CRUD test examples

Both specs were run against an `orders` resource generated by `backend-feature-slice` + `frontend-feature-pages`, with fields `title:string, notes:string?, quantity:int, priceCents:int, paid:boolean, dueAt:date?`. Adapt the names, fields and labels to the resource under test. Keep the structure.

To validate API responses against contracts, add `"@aure/contracts": "workspace:*"` to `apps/e2e` devDependencies once the first resource exists.

## API: `tests/api/orders.spec.ts`

```ts
import { OrderListSchema, OrderSchema } from "@aure/contracts";
import { expect, test } from "@playwright/test";

const unique = (label: string): string => `${label} ${crypto.randomUUID().slice(0, 8)}`;

test.describe("orders API", () => {
  test("creates, reads, updates and deletes an order", async ({ request }) => {
    const title = unique("api order");
    const created = await request.post("/api/orders", {
      data: { title, quantity: 2, priceCents: 500, paid: false },
    });
    expect(created.status()).toBe(201);
    // Parsing with the contract schema proves the response shape, not just the status.
    const order = OrderSchema.parse(await created.json());
    expect(order).toMatchObject({ title, notes: null, quantity: 2, paid: false });

    const fetched = await request.get(`/api/orders/${order.id}`);
    expect(fetched.status()).toBe(200);
    expect(OrderSchema.parse(await fetched.json())).toEqual(order);

    const updated = await request.patch(`/api/orders/${order.id}`, { data: { notes: "rush", paid: true } });
    expect(updated.status()).toBe(200);
    expect(OrderSchema.parse(await updated.json())).toMatchObject({ title, notes: "rush", paid: true });

    const deleted = await request.delete(`/api/orders/${order.id}`);
    expect(deleted.status()).toBe(204);
    expect((await request.get(`/api/orders/${order.id}`)).status()).toBe(404);
  });

  test("lists orders newest first with paging", async ({ request }) => {
    const ids: string[] = [];
    for (const label of ["first", "second"]) {
      const response = await request.post("/api/orders", {
        data: { title: unique(label), quantity: 1, priceCents: 100, paid: false },
      });
      ids.push(OrderSchema.parse(await response.json()).id);
    }
    const page = OrderListSchema.parse(await (await request.get("/api/orders?limit=100")).json());
    expect(page.limit).toBe(100);
    const positions = ids.map((id) => page.items.findIndex((item) => item.id === id));
    // Other tests create orders in parallel, so check relative order, not exact contents.
    expect(positions.every((position) => position >= 0)).toBe(true);
    expect(positions[1]).toBeLessThan(positions[0] ?? -1);
  });

  test("rejects invalid bodies with field-level issues", async ({ request }) => {
    const response = await request.post("/api/orders", { data: { title: "x", quantity: 1.5, paid: "yes" } });
    expect(response.status()).toBe(400);
    const body = (await response.json()) as { issues: { path: string }[] };
    expect(body.issues.map((issue) => issue.path)).toEqual(expect.arrayContaining(["quantity", "priceCents", "paid"]));
  });

  test("rejects an empty patch, a malformed id and an unknown id", async ({ request }) => {
    expect((await request.get("/api/orders/not-a-uuid")).status()).toBe(400);
    expect((await request.get(`/api/orders/${crypto.randomUUID()}`)).status()).toBe(404);
    expect((await request.patch(`/api/orders/${crypto.randomUUID()}`, { data: {} })).status()).toBe(400);
  });
});
```

## Web: `tests/web/orders.spec.ts`

```ts
import { expect, gotoHydrated, test } from "./fixtures";

test.describe("orders pages", () => {
  test("creates, edits and deletes an order through the UI", async ({ page }) => {
    const title = `ui order ${crypto.randomUUID().slice(0, 8)}`;
    await gotoHydrated(page, "/orders/new");

    // Client-side validation from the contract schema.
    await page.getByRole("button", { name: "Create order" }).click();
    await expect(page.getByLabel("Quantity")).toHaveAttribute("aria-invalid", "true");

    await page.getByLabel("Title").fill(title);
    await page.getByLabel("Quantity").fill("3");
    await page.getByLabel("Price cents").fill("1500");
    await page.getByLabel("Paid").check();
    await page.getByLabel("Due at (UTC)").fill("2026-10-01T09:30");
    await page.getByRole("button", { name: "Create order" }).click();

    await expect(page).toHaveURL(/\/orders\/[0-9a-f-]{36}$/);
    await expect(page.getByRole("heading", { level: 1, name: title })).toBeVisible();
    await expect(page).toHaveTitle(`${title} · aure`);
    await expect(page.locator('dt:text-is("Due at") + dd')).toHaveText("Oct 1, 2026, 9:30 AM UTC");

    await page.getByRole("link", { name: "Edit" }).click();
    await expect(page.getByLabel("Quantity")).toHaveValue("3");
    await page.getByLabel("Quantity").fill("5");
    await page.getByRole("button", { name: "Save changes" }).click();
    await expect(page.locator('dt:text-is("Quantity") + dd')).toHaveText("5");

    await page.getByRole("button", { name: "Delete" }).click();
    await page.getByRole("button", { name: "Yes, delete" }).click();
    await expect(page).toHaveURL(/\/orders$/);
    await expect(page.getByRole("link", { name: title })).toHaveCount(0);
  });

  test.describe("unknown order", () => {
    // The document itself is a 404, which the browser logs as a console error.
    test.use({ allowedConsoleErrors: [/status of 404/] });

    test("shows the not-found page", async ({ page }) => {
      const response = await gotoHydrated(page, `/orders/${crypto.randomUUID()}`);
      expect(response?.status()).toBe(404);
      await expect(page.getByRole("heading", { name: "Not found" })).toBeVisible();
    });
  });
});
```
